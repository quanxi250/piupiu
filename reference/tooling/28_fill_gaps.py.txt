"""补齐角色卡：扫全部页拿 id → 找出本地缺失 → 只补缺失的"""
import gzip, json, os, re, sys, time, urllib.request

BASE = "<TARGET_URL>"
TOKEN = "YjDjtyN-gvr5tAXwxXklmOCsRRVR9ssjExoQpAVJ8Pg"
DEV = "b6b406199e5b3a1d"
DST = r"E:\gzq\piupiu-market\角色卡_JSON"
RAW = r"E:\gzq\piupiu-market\_原始响应\roleurl"
UA = "Mozilla/5.0 (Linux; Android 13) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Mobile Safari/537.36"

os.makedirs(DST, exist_ok=True)
os.makedirs(RAW, exist_ok=True)


def post(body, retries=4):
    body = dict(body)
    body["access_token"] = TOKEN
    body["device_id"] = DEV
    data = json.dumps(body).encode()
    for a in range(retries):
        try:
            req = urllib.request.Request(BASE, data=data, headers={
                "Content-Type": "application/json",
                "User-Agent": UA,
                "Accept": "application/json",
            })
            with urllib.request.urlopen(req, timeout=30) as r:
                txt = r.read().decode("utf-8", "replace")
            if "Access denied" in txt:
                time.sleep(15 + a * 10)
                continue
            return txt
        except Exception as e:
            time.sleep(5 + a * 5)
    return None


def local_ids():
    ids = set()
    for f in os.listdir(DST):
        m = re.search(r"__(\d+)\.json$", f)
        if m:
            ids.add(int(m.group(1)))
    return ids


def safe(n, fb):
    n = re.sub(r'[\\/:*?"<>|\r\n\t]', "_", (n or "").strip())
    return (re.sub(r"\s+", " ", n).strip(" .")[:80] or fb)


print("[1] 扫描全部页面收集 id ...", flush=True)
seen = {}
page = 1
total_pages = 294
while page <= total_pages:
    txt = post({"action": "list", "page": page, "page_size": 50, "sort": "",
                "tag": "", "search": "", "random_seed": 0})
    if txt is None:
        print(f"  第 {page} 页失败，跳过", flush=True)
        page += 1
        continue
    try:
        j = json.loads(txt)
    except Exception:
        print(f"  第 {page} 页解析失败", flush=True)
        page += 1
        continue
    d = j.get("data") or {}
    total_pages = d.get("total_pages") or total_pages
    for it in (d.get("items") or []):
        i = it.get("id")
        if i:
            seen[int(i)] = it
    if page % 30 == 0 or page == total_pages:
        print(f"  页 {page}/{total_pages}  累计 id {len(seen)}", flush=True)
    page += 1
    time.sleep(1.1)

print(f"[2] 全站 id 数: {len(seen)}", flush=True)
have = local_ids()
missing = sorted(set(seen) - have)
print(f"    本地 {len(have)}  缺失 {len(missing)}", flush=True)

if not missing:
    print("无需补漏")
    sys.exit(0)

print("[3] 补下载缺失卡片 ...", flush=True)
ok = fail = 0
for idx, cid in enumerate(missing, 1):
    it = seen[cid]
    txt = post({"action": "download", "id": cid})
    if not txt:
        fail += 1
        continue
    try:
        j = json.loads(txt)
    except Exception:
        fail += 1
        continue
    d = j.get("data") or {}
    url = d.get("data_url")
    if not url:
        fail += 1
        continue
    try:
        req = urllib.request.Request(url, headers={"User-Agent": UA})
        with urllib.request.urlopen(req, timeout=40) as r:
            raw = r.read()
        if url.endswith(".gz"):
            raw = gzip.decompress(raw)
        obj = json.loads(raw.decode("utf-8"))
        name = safe((obj.get("data") or {}).get("name") or it.get("name"), f"card_{cid}")
        path = os.path.join(DST, f"{name}__{cid}.json")
        if not os.path.exists(path):
            open(path, "wb").write(raw)
        # 记录 url 供后续换封面
        rec = os.path.join(RAW, f"_roleurl_all_{cid:06d}.json")
        if not os.path.exists(rec):
            open(rec, "w", encoding="utf-8").write(json.dumps(
                {"id": cid, "n": it.get("name"), "url": url, "cover": d.get("cover_url"), "ok": 1},
                ensure_ascii=False))
        ok += 1
    except Exception as e:
        fail += 1
    if idx % 25 == 0 or idx == len(missing):
        print(f"  补 {idx}/{len(missing)}  成功 {ok} 失败 {fail}", flush=True)
    time.sleep(1.1)

print(f"[完成] 成功 {ok} 失败 {fail}", flush=True)
